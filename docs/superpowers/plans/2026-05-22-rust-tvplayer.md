# Rust TV Player Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `tvplayer` — a clean Rust IPTV player on libmpv + egui + wgpu — and ship a working portable .zip.

**Architecture:** Single-binary Rust crate. libmpv (hand-written FFI) handles AV, codecs, subs, sync. egui + eframe (wgpu backend) handles overlays. Tokio runtime owns the PlayerActor + portal HTTP. mpv software-render → RGBA buffer → wgpu texture → egui composition.

**Tech Stack:** Rust stable, MSVC toolchain, egui/eframe, tokio, reqwest, serde, libmpv-2 (vendored DLL). Windows-only v1.

**Spec:** `docs/superpowers/specs/2026-05-22-rust-tvplayer-design.md`

```powershell
# Idempotent: skips work if files already present.

$ErrorActionPreference = 'Stop'
$here = Split-Path -Parent $MyInvocation.MyCommand.Path
$dest = Join-Path $here 'libmpv'

New-Item -ItemType Directory -Path $dest -Force | Out-Null
New-Item -ItemType Directory -Path (Join-Path $dest 'include\mpv') -Force | Out-Null

$dll = Join-Path $dest 'libmpv-2.dll'
$lib = Join-Path $dest 'libmpv.lib'

if ((Test-Path $dll) -and (Test-Path $lib) -and (Test-Path (Join-Path $dest 'include\mpv\client.h'))) {
    Write-Host "libmpv already vendored at $dest"
    exit 0
}

# Stable mpv-dev release. shinchiro builds are the de-facto Windows libmpv distribution.
$url = 'https://github.com/shinchiro/mpv-winbuild-cmake/releases/download/20240609/mpv-dev-x86_64-20240609-git-571d4c8.7z'
$tmpZip = Join-Path $env:TEMP 'mpv-dev.7z'
$tmpExtract = Join-Path $env:TEMP 'mpv-dev'

Write-Host "Downloading $url"
Invoke-WebRequest -Uri $url -OutFile $tmpZip

if (Test-Path $tmpExtract) { Remove-Item $tmpExtract -Recurse -Force }
New-Item -ItemType Directory -Path $tmpExtract -Force | Out-Null

# Use 7-Zip if available, else fallback to tar (Windows 10+).
$sevenZip = Get-Command 7z.exe -ErrorAction SilentlyContinue
if ($sevenZip) {
    & 7z.exe x $tmpZip -o"$tmpExtract" -y | Out-Null
} else {
    # Fallback: tar can read 7z via libarchive on recent Windows 10+
    tar -xf $tmpZip -C $tmpExtract
}

Copy-Item (Join-Path $tmpExtract 'libmpv-2.dll') $dll -Force
Copy-Item (Join-Path $tmpExtract 'include\mpv\client.h') (Join-Path $dest 'include\mpv\client.h') -Force
Copy-Item (Join-Path $tmpExtract 'include\mpv\render.h') (Join-Path $dest 'include\mpv\render.h') -Force

Copy-Item (Join-Path $tmpExtract 'include\mpv\stream_cb.h') (Join-Path $dest 'include\mpv\stream_cb.h') -Force -ErrorAction SilentlyContinue

# Generate import library from the DLL using dumpbin + lib (MSVC toolchain).
$vsDevCmd = Get-ChildItem 'C:\Program Files\Microsoft Visual Studio' -Recurse -Filter VsDevCmd.bat -ErrorAction SilentlyContinue | Select-Object -First 1
if (-not $vsDevCmd) {
    $vsDevCmd = Get-ChildItem 'C:\Program Files (x86)\Microsoft Visual Studio' -Recurse -Filter VsDevCmd.bat -ErrorAction SilentlyContinue | Select-Object -First 1
}
if (-not $vsDevCmd) { throw 'VsDevCmd.bat not found; install VS Build Tools with C++ workload.' }

$defFile = Join-Path $dest 'libmpv-2.def'
$exports = & cmd /c "`"$($vsDevCmd.FullName)`" -arch=x64 -no_logo >nul && dumpbin /exports `"$dll`""
$lines = $exports -split "`n"
$names = @()
$capture = $false
foreach ($line in $lines) {
    if ($line -match '^\s+ordinal\s+hint\s+RVA\s+name') { $capture = $true; continue }
    if (-not $capture) { continue }
    if ($line -match '^\s*Summary') { break }
    if ($line -match '^\s*\d+\s+[0-9A-F]+\s+[0-9A-F]+\s+(\S+)') { $names += $matches[1] }

"LIBRARY libmpv-2.dll`nEXPORTS" + "`n" + (($names | ForEach-Object { "  $_" }) -join "`n") | Set-Content $defFile -Encoding ASCII
```

```rust
use clap::Parser;
use thiserror::Error;

#[derive(Parser, Debug)]
#[command(name = "tvplayer", version, about = "Lightweight IPTV player")]
pub struct Cli {
    /// Xtream Codes portal: user:pass@host[:port]
    #[arg(long)]
    pub xtream: Option<String>,

    /// Direct stream URL (HLS/HTTP/etc.) — bypasses portal
    pub url: Option<String>,

    /// Run smoke selftest and exit
    #[arg(long, hide = true)]
    pub selftest: bool,
}

#[derive(Debug, PartialEq, Eq, Clone)]
pub struct XtreamCreds {
    pub username: String,
    pub password: String,
    pub host: String,
    pub port: u16,
}

#[derive(Debug, Error)]
pub enum CredsError {
    #[error("missing '@' in creds")]
    NoAt,
    #[error("missing ':' separator in user:pass")]
    NoUserPass,
    #[error("invalid port")]
    BadPort,
}

pub fn parse_xtream_creds(s: &str) -> Result<XtreamCreds, CredsError> {
    let (userpass, hostport) = s.split_once('@').ok_or(CredsError::NoAt)?;
    let (user, pass) = userpass.split_once(':').ok_or(CredsError::NoUserPass)?;
    let (host, port) = match hostport.split_once(':') {
        Some((h, p)) => (h, p.parse::<u16>().map_err(|_| CredsError::BadPort)?),
        None => (hostport, 80),
    };
    Ok(XtreamCreds {
        username: user.into(),
        password: pass.into(),
        host: host.into(),
        port,
    })
}
```

```rust
use std::path::{Path, PathBuf};

pub struct Storage {
    root: PathBuf,
}

impl Storage {
    /// Standard location: %APPDATA%\tvplayer\ on Windows.
    pub fn standard() -> anyhow::Result<Self> {
        let proj = directories::ProjectDirs::from("", "", "tvplayer")
            .ok_or_else(|| anyhow::anyhow!("no project dir"))?;
        Ok(Self { root: proj.config_dir().to_path_buf() })
    }

    /// Override root — used in tests.
    pub fn with_root(root: PathBuf) -> Self {
        Self { root }
    }

    pub fn config_dir(&self) -> &Path { &self.root }

    pub fn ensure_config_dir(&self) -> std::io::Result<PathBuf> {
        std::fs::create_dir_all(&self.root)?;
        Ok(self.root.clone())
    }

    pub fn favorites_path(&self) -> PathBuf { self.root.join("favorites.json") }
    pub fn config_path(&self) -> PathBuf { self.root.join("config.json") }
}
```

```rust
use serde::{Deserialize, Serialize};
use std::path::Path;

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct Entry {
    pub stream_id: i64,
    pub name: String,
}

#[derive(Debug, Default, Clone, Serialize, Deserialize)]
pub struct Favorites {
    entries: Vec<Entry>,
}

impl Favorites {
    pub fn contains(&self, stream_id: i64) -> bool {
        self.entries.iter().any(|e| e.stream_id == stream_id)
    }

    pub fn toggle(&mut self, stream_id: i64, name: &str) {
        if let Some(pos) = self.entries.iter().position(|e| e.stream_id == stream_id) {
            self.entries.remove(pos);
        } else {
            self.entries.push(Entry { stream_id, name: name.to_owned() });
        }
    }

    pub fn remove(&mut self, stream_id: i64) {
        self.entries.retain(|e| e.stream_id != stream_id);
    }

    pub fn iter(&self) -> std::slice::Iter<'_, Entry> { self.entries.iter() }

    pub fn save(&self, path: &Path) -> anyhow::Result<()> {
        if let Some(p) = path.parent() { std::fs::create_dir_all(p)?; }
        let s = serde_json::to_string_pretty(self)?;
        std::fs::write(path, s)?;
        Ok(())
    }

    pub fn load(path: &Path) -> anyhow::Result<Self> {
        if !path.exists() { return Ok(Self::default()); }
        let s = std::fs::read_to_string(path)?;
        Ok(serde_json::from_str(&s)?)
    }
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq, Copy)]
pub enum ItemKind { Live, Movie, Series }

#[derive(Debug, Clone)]
pub struct SearchItem {
    pub id: i64,
    pub name: String,
    pub kind: ItemKind,
}

pub fn rank<'a>(query: &str, items: &'a [SearchItem]) -> Vec<&'a SearchItem> {
    let q = query.trim().to_lowercase();
    if q.is_empty() { return Vec::new(); }

    let mut hits: Vec<&SearchItem> = items.iter()
        .filter(|it| it.name.to_lowercase().contains(&q))
        .collect();

    hits.sort_by(|a, b| {
        // Kind priority first (Live < Movie < Series numerically = best first)
        let ak = kind_rank(a.kind);
        let bk = kind_rank(b.kind);
        ak.cmp(&bk)
            // Then shorter name first (closer match)
            .then_with(|| a.name.len().cmp(&b.name.len()))
            // Then alphabetic for stability
            .then_with(|| a.name.cmp(&b.name))
    });

    hits.truncate(12);
    hits
}

fn kind_rank(k: ItemKind) -> u8 {
    match k {
        ItemKind::Live => 0,
        ItemKind::Movie => 1,
        ItemKind::Series => 2,
    }
}
```

```rust
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct EpgEntry {

    pub start: DateTime<Utc>,
    pub end: DateTime<Utc>,

#[derive(Debug, Default, Clone)]
pub struct Epg {
    entries: Vec<EpgEntry>,  // assumed sorted by start ascending
}

impl Epg {
    pub fn new(mut entries: Vec<EpgEntry>) -> Self {
        entries.sort_by_key(|e| e.start);
        Self { entries }
    }

    pub fn entries(&self) -> &[EpgEntry] { &self.entries }

    pub fn current_at(&self, t: DateTime<Utc>) -> Option<&EpgEntry> {
        self.entries.iter().find(|e| e.start <= t && t < e.end)
    }

    pub fn next_at(&self, t: DateTime<Utc>) -> Option<&EpgEntry> {
        self.entries.iter().find(|e| e.start > t)
    }
}
```

```rust
use crate::args::XtreamCreds;
use crate::epg::{Epg, EpgEntry};
use super::{Portal, PortalError, types::*};
use chrono::{TimeZone, Utc};
use serde::Deserialize;

#[derive(Debug, Clone)]
pub struct XtreamPortal {
    pub creds: XtreamCreds,
    client: reqwest::Client,
}

impl XtreamPortal {
    pub fn new(creds: XtreamCreds) -> Self {
        let client = reqwest::Client::builder()
            .user_agent("tvplayer/0.1")
            .timeout(std::time::Duration::from_secs(15))
            .build()
            .expect("build reqwest client");
        Self { creds, client }
    }

    fn api_url(&self, params: &[(&str, &str)]) -> String {
        let mut q: Vec<(&str, &str)> = vec![
            ("username", &self.creds.username),
            ("password", &self.creds.password),
        ];
        q.extend_from_slice(params);
        let qs: String = q.iter().map(|(k, v)| format!("{}={}", k, urlencoding::encode(v))).collect::<Vec<_>>().join("&");
        format!("http://{}:{}/player_api.php?{}", self.creds.host, self.creds.port, qs)
    }

    fn base(&self) -> String { format!("http://{}:{}", self.creds.host, self.creds.port) }
}

#[derive(Deserialize)]
struct EpgWrapper {
    epg_listings: Vec<EpgRaw>,
}

#[derive(Deserialize)]
struct EpgRaw {
    #[serde(default)]
    title: String,
    #[serde(default)]
    start_timestamp: String,
    #[serde(default)]
    stop_timestamp: String,
}

fn b64_decode(s: &str) -> String {
    use base64::Engine;
    base64::engine::general_purpose::STANDARD.decode(s)
        .ok()
        .and_then(|b| String::from_utf8(b).ok())
        .unwrap_or_else(|| s.to_owned())
}

#[async_trait::async_trait]
impl Portal for XtreamPortal {
    async fn fetch_catalog(&self) -> Result<Catalog, PortalError> {
        let live: Vec<LiveChannel> = self.client.get(self.api_url(&[("action", "get_live_streams")])).send().await?.json().await?;
        let movies: Vec<Movie> = self.client.get(self.api_url(&[("action", "get_vod_streams")])).send().await?.json().await?;
        let series: Vec<Series> = self.client.get(self.api_url(&[("action", "get_series")])).send().await?.json().await?;
        Ok(Catalog { live, movies, series })
    }

    async fn fetch_epg(&self, stream_id: i64) -> Result<Epg, PortalError> {
        let sid = stream_id.to_string();
        let url = self.api_url(&[("action", "get_short_epg"), ("stream_id", &sid), ("limit", "100")]);
        let txt = self.client.get(url).send().await?.text().await?;
        let wrap: EpgWrapper = serde_json::from_str(&txt).map_err(|e| PortalError::Shape(e.to_string()))?;
        let entries: Vec<EpgEntry> = wrap.epg_listings.into_iter().filter_map(|r| {
            let start = r.start_timestamp.parse::<i64>().ok()?;
            let end   = r.stop_timestamp.parse::<i64>().ok()?;
            Some(EpgEntry {
                title: b64_decode(&r.title),
                start: Utc.timestamp_opt(start, 0).single()?,
                end:   Utc.timestamp_opt(end, 0).single()?,
            })
        }).collect();
        Ok(Epg::new(entries))
    }

    async fn fetch_series_episodes(&self, series_id: i64) -> Result<Vec<Episode>, PortalError> {
        let sid = series_id.to_string();
        let url = self.api_url(&[("action", "get_series_info"), ("series_id", &sid)]);
        let v: serde_json::Value = self.client.get(url).send().await?.json().await?;
        let mut out = Vec::new();
        if let Some(seasons) = v.get("episodes").and_then(|x| x.as_object()) {
            for (season_str, eps) in seasons.iter() {
                let season: i64 = season_str.parse().unwrap_or(0);
                if let Some(arr) = eps.as_array() {
                    for ep in arr {
                        let id = ep.get("id").and_then(|x| x.as_str()).unwrap_or("").to_owned();
                        let title = ep.get("title").and_then(|x| x.as_str()).unwrap_or("").to_owned();
                        let ext = ep.get("container_extension").and_then(|x| x.as_str()).unwrap_or("mp4").to_owned();
                        let n = ep.get("episode_num").and_then(|x| x.as_i64()).unwrap_or(0);
                        out.push(Episode { id, title, container_extension: ext, season, episode_num: n });
                    }
                }
            }
        }
        Ok(out)
    }

    fn live_stream_url(&self, stream_id: i64) -> String {
        format!("{}/live/{}/{}/{}.m3u8", self.base(), self.creds.username, self.creds.password, stream_id)
    }

    fn movie_stream_url(&self, stream_id: i64, container_ext: &str) -> String {
        format!("{}/movie/{}/{}/{}.{}", self.base(), self.creds.username, self.creds.password, stream_id, container_ext)
    }

    fn series_stream_url(&self, episode_id: &str, container_ext: &str) -> String {
        format!("{}/series/{}/{}/{}.{}", self.base(), self.creds.username, self.creds.password, episode_id, container_ext)
    }
}
```

```rust
#[derive(Debug, Error)]
pub enum MpvError {
    #[error("mpv error {0}: {1}")]
    Code(i32, String),
    #[error("mpv handle is null (create failed)")]
    NullHandle,

pub struct Mpv {

unsafe impl Send for Mpv {}
unsafe impl Sync for Mpv {}

impl Mpv {
    pub fn new() -> Result<Self, MpvError> {
        let h = unsafe { sys::mpv_create() };
        if h.is_null() { return Err(MpvError::NullHandle); }

        let me = Self { handle: h };

        // Reasonable defaults BEFORE initialize:
        me.set_option("config", "no")?;
        me.set_option("audio-display", "no")?;
        me.set_option("cache-secs", "10")?;
        me.set_option("hwdec", "auto-safe")?;
        me.set_option("vo", "libmpv")?;
        // input handling done in Rust — disable mpv's built-in keyboard:
        me.set_option("input-default-bindings", "no")?;
        me.set_option("input-vo-keyboard", "no")?;
        me.set_option("osc", "no")?;
        // Subs always on by default when present:
        me.set_option("sid", "auto")?;
        me.set_option("sub-auto", "fuzzy")?;

        check(unsafe { sys::mpv_initialize(h) }, h)?;

        // Verbose-ish logging routed to our tracing layer:
        let lvl = CString::new("info").unwrap();
        unsafe { sys::mpv_request_log_messages(h, lvl.as_ptr()); }

        Ok(me)
    }

    pub fn raw(&self) -> *mut sys::mpv_handle { self.handle }

    pub fn set_option(&self, name: &str, value: &str) -> Result<(), MpvError> {
        let n = CString::new(name).unwrap();
        let v = CString::new(value).unwrap();
        check(unsafe { sys::mpv_set_option_string(self.handle, n.as_ptr(), v.as_ptr()) }, self.handle)
    }

    pub fn set_property(&self, name: &str, value: &str) -> Result<(), MpvError> {
        let n = CString::new(name).unwrap();
        let v = CString::new(value).unwrap();
        check(unsafe { sys::mpv_set_property_string(self.handle, n.as_ptr(), v.as_ptr()) }, self.handle)
    }

    pub fn get_property_string(&self, name: &str) -> Option<String> {
        let n = CString::new(name).unwrap();
        unsafe {
            let p = sys::mpv_get_property_string(self.handle, n.as_ptr());
            if p.is_null() { return None; }
            let s = CStr::from_ptr(p).to_string_lossy().into_owned();
            sys::mpv_free(p as *mut c_void);
            Some(s)
        }
    }

    pub fn get_property_f64(&self, name: &str) -> Option<f64> {
        let n = CString::new(name).unwrap();
        let mut out: f64 = 0.0;
        let rc = unsafe { sys::mpv_get_property(self.handle, n.as_ptr(), sys::MPV_FORMAT_DOUBLE, &mut out as *mut _ as *mut c_void) };
        if rc < 0 { None } else { Some(out) }
    }

    pub fn command(&self, args: &[&str]) -> Result<(), MpvError> {
        let cstrs: Vec<CString> = args.iter().map(|a| CString::new(*a).unwrap()).collect();
        let mut ptrs: Vec<*const std::os::raw::c_char> = cstrs.iter().map(|c| c.as_ptr()).collect();
        ptrs.push(ptr::null());
        check(unsafe { sys::mpv_command(self.handle, ptrs.as_mut_ptr()) }, self.handle)
    }

    pub fn observe_property(&self, name: &str, format: i32, userdata: u64) -> Result<(), MpvError> {
        let n = CString::new(name).unwrap();
        check(unsafe { sys::mpv_observe_property(self.handle, userdata, n.as_ptr(), format) }, self.handle)
    }

    pub fn wait_event(&self, timeout_s: f64) -> *mut sys::mpv_event {
        unsafe { sys::mpv_wait_event(self.handle, timeout_s) }

        unsafe { sys::mpv_set_wakeup_callback(self.handle, cb, ud); }
    }
}

impl Drop for Mpv {

        if !self.handle.is_null() {
            unsafe { sys::mpv_terminate_destroy(self.handle); }
            self.handle = ptr::null_mut();
        }
    }
}

fn check(rc: i32, h: *mut sys::mpv_handle) -> Result<(), MpvError> {
    if rc >= 0 { return Ok(()); }
    let msg = unsafe {
        let p = sys::mpv_error_string(rc);
        if p.is_null() { "unknown".into() } else { CStr::from_ptr(p).to_string_lossy().into_owned() }
    };
    let _ = h; // currently we don't use the handle for error lookup
    Err(MpvError::Code(rc, msg))
}
```

```rust
use super::mpv_sys as sys;
use super::mpv::Mpv;
use parking_lot::Mutex;
use std::ffi::CString;
use std::os::raw::c_void;
use std::ptr;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct RgbaFrame {
    pub w: u32,
    pub h: u32,
    pub data: Vec<u8>,  // length = w*h*4, BGRA0
    pub version: u64,
}

impl Default for RgbaFrame {
    fn default() -> Self { Self { w: 0, h: 0, data: Vec::new(), version: 0 } }
}

/// Single-slot shared frame. Render task writes, UI reads.
/// "Triple buffer" emergent behavior: rendering happens behind the lock,
/// UI clones data only when there's a new version.
pub struct FrameBus {
    inner: Mutex<RgbaFrame>,
    on_new_frame: Mutex<Option<Box<dyn Fn() + Send + Sync>>>,
}

impl FrameBus {
    pub fn new() -> Self {
        Self { inner: Mutex::new(RgbaFrame::default()), on_new_frame: Mutex::new(None) }
    }

    pub fn set_new_frame_callback(&self, cb: impl Fn() + Send + Sync + 'static) {
        *self.on_new_frame.lock() = Some(Box::new(cb));
    }

    pub fn read(&self) -> RgbaFrame {
        self.inner.lock().clone()
    }

    fn write(&self, w: u32, h: u32, data: Vec<u8>) {
        {
            let mut g = self.inner.lock();
            g.w = w; g.h = h; g.data = data;
            g.version = g.version.wrapping_add(1);
        }
        if let Some(cb) = self.on_new_frame.lock().as_ref() { cb(); }
    }
}

pub struct RenderCtx {
    raw: *mut sys::mpv_render_context,
    frames: Arc<FrameBus>,
    width: u32,
    height: u32,
    buf: Vec<u8>,
}

unsafe impl Send for RenderCtx {}
unsafe impl Sync for RenderCtx {}

impl RenderCtx {
    pub fn new(mpv: &Mpv, frames: Arc<FrameBus>, init_w: u32, init_h: u32) -> Result<Self, super::mpv::MpvError> {
        let mut sw_type = CString::new("sw").unwrap();
        let api_type_key = CString::new("api-type").unwrap();
        let _ = (sw_type, api_type_key); // (suppress unused if reorg)

        let api_type = sys::MPV_RENDER_API_TYPE_SW;

        // Build params array.
        let mut params: [sys::mpv_render_param; 2] = [
            sys::mpv_render_param {
                type_: sys::MPV_RENDER_PARAM_API_TYPE,
                data: &api_type as *const _ as *mut c_void,
            },
            sys::mpv_render_param {
                type_: sys::MPV_RENDER_PARAM_INVALID,
                data: ptr::null_mut(),
            },
        ];

        let mut ctx: *mut sys::mpv_render_context = ptr::null_mut();
        let rc = unsafe { sys::mpv_render_context_create(&mut ctx as *mut _, mpv.raw(), params.as_mut_ptr()) };
        if rc < 0 || ctx.is_null() {
            return Err(super::mpv::MpvError::Code(rc, "mpv_render_context_create failed".into()));
        }

        let buf = vec![0u8; (init_w as usize) * (init_h as usize) * 4];
        Ok(Self { raw: ctx, frames, width: init_w, height: init_h, buf })
    }

    pub fn resize(&mut self, w: u32, h: u32) {
        if w == self.width && h == self.height { return; }
        self.width = w.max(1);
        self.height = h.max(1);
        self.buf = vec![0u8; (self.width as usize) * (self.height as usize) * 4];
    }

    /// Synchronously render the current mpv frame into our buffer and publish.
    pub fn render_into_frames(&mut self) -> Result<(), super::mpv::MpvError> {
        // SW params: format=bgra0, w/h, stride=w*4, pointer=buf.
        let format = CString::new("bgra0").unwrap();
        let mut size_arr: [i32; 2] = [self.width as i32, self.height as i32];
        let mut stride: usize = (self.width as usize) * 4;

        let mut params: [sys::mpv_render_param; 5] = [
            sys::mpv_render_param { type_: sys::MPV_RENDER_PARAM_SW_SIZE,    data: size_arr.as_mut_ptr() as *mut c_void },
            sys::mpv_render_param { type_: sys::MPV_RENDER_PARAM_SW_FORMAT,  data: format.as_ptr() as *mut c_void },
            sys::mpv_render_param { type_: sys::MPV_RENDER_PARAM_SW_STRIDE,  data: &mut stride as *mut _ as *mut c_void },
            sys::mpv_render_param { type_: sys::MPV_RENDER_PARAM_SW_POINTER, data: self.buf.as_mut_ptr() as *mut c_void },
            sys::mpv_render_param { type_: sys::MPV_RENDER_PARAM_INVALID,    data: ptr::null_mut() },
        ];
        let rc = unsafe { sys::mpv_render_context_render(self.raw, params.as_mut_ptr()) };
        if rc < 0 {
            return Err(super::mpv::MpvError::Code(rc, "mpv_render_context_render failed".into()));
        }
        // Publish a clone — UI then reads independently.
        self.frames.write(self.width, self.height, self.buf.clone());
        Ok(())
    }

    /// Returns true if mpv signals a new frame is ready.
    pub fn has_new_frame(&self) -> bool {
        let flags = unsafe { sys::mpv_render_context_update(self.raw) };
        flags & sys::MPV_RENDER_UPDATE_FRAME != 0
    }
}

impl Drop for RenderCtx {
    fn drop(&mut self) {
        if !self.raw.is_null() {
            unsafe { sys::mpv_render_context_free(self.raw); }
            self.raw = ptr::null_mut();
        }
    }
}
```

```rust
        sys::MPV_EVENT_PLAYBACK_RESTART  => Some(Event::PlaybackStarted),
        sys::MPV_EVENT_END_FILE          => Some(Event::EndOfFile { reason: end_file_reason(evt) }),
        sys::MPV_EVENT_LOG_MESSAGE       => {

            None
        },
        _ => None,
    }

unsafe fn end_file_reason(evt: *mut sys::mpv_event) -> String {
    let data = (*evt).data;
    if data.is_null() { return "unknown".into(); }
    // struct mpv_event_end_file { reason: int; error: int; ... } — we read the first int.
    let reason_code = *(data as *const i32);
    match reason_code {
        0 => "eof".into(),
        1 => "stop".into(),
        2 => "quit".into(),
        3 => "error".into(),
        4 => "redirect".into(),
        _ => format!("reason={}", reason_code),
    }
}

pub unsafe fn property_change(evt: *mut sys::mpv_event) -> Option<(String, String)> {
    let data = (*evt).data as *mut sys::mpv_event_property;
    if data.is_null() { return None; }
    let name = CStr::from_ptr((*data).name).to_string_lossy().into_owned();
    let val = if (*data).format == sys::MPV_FORMAT_STRING && !(*data).data.is_null() {
        let p = *((*data).data as *const *const std::os::raw::c_char);
        if p.is_null() { String::new() } else { CStr::from_ptr(p).to_string_lossy().into_owned() }
```

```rust
pub mod mpv_sys;
pub mod mpv;
pub mod render;
pub mod events;

pub use mpv::{Mpv, MpvError};
pub use render::{FrameBus, RgbaFrame};

use std::sync::Arc;
use std::thread;
use std::time::Duration;
use tokio::sync::mpsc::{UnboundedReceiver, UnboundedSender, unbounded_channel};
use tracing::{debug, error, info, warn};

#[derive(Debug)]
pub enum Cmd {
    LoadUrl(String),
    Pause(bool),
    SeekRelative(f64),
    SetAudioTrack(i64),
    SetSubtitleTrack(i64),  // -1 disables
    SetWindowSize(u32, u32),
    Quit,
}

#[derive(Debug, Clone)]
pub enum Event {
    FileLoaded,
    PlaybackStarted,
    EndOfFile { reason: String },
    Error { msg: String },
    PropertyChanged { name: String, value: String },
}

pub struct PlayerHandle {
    pub cmd_tx: UnboundedSender<Cmd>,
    pub evt_rx: parking_lot::Mutex<UnboundedReceiver<Event>>,
    pub frames: Arc<FrameBus>,
}

pub fn spawn(initial_w: u32, initial_h: u32) -> anyhow::Result<PlayerHandle> {
    let (cmd_tx, mut cmd_rx) = unbounded_channel::<Cmd>();
    let (evt_tx, evt_rx) = unbounded_channel::<Event>();
    let frames = Arc::new(FrameBus::new());
    let frames_for_thread = frames.clone();

    thread::Builder::new().name("mpv-player".into()).spawn(move || {
        let mpv = match Mpv::new() {
            Ok(m) => m,
            Err(e) => { let _ = evt_tx.send(Event::Error { msg: e.to_string() }); return; }
        };
        let mut rctx = match render::RenderCtx::new(&mpv, frames_for_thread, initial_w.max(1), initial_h.max(1)) {
            Ok(r) => r,
            Err(e) => { let _ = evt_tx.send(Event::Error { msg: e.to_string() }); return; }
        };

        info!("mpv player thread started");

        loop {
            // Drain incoming commands (non-blocking).
            loop {
                match cmd_rx.try_recv() {
                    Ok(Cmd::LoadUrl(url)) => {
                        info!("loadfile {}", url);
                        if let Err(e) = mpv.command(&["loadfile", &url]) {
                            warn!("loadfile failed: {}", e);
                            let _ = evt_tx.send(Event::Error { msg: e.to_string() });
                        }
                    },
                    Ok(Cmd::Pause(p)) => { let _ = mpv.set_property("pause", if p { "yes" } else { "no" }); },
                    Ok(Cmd::SeekRelative(s)) => { let _ = mpv.command(&["seek", &s.to_string(), "relative"]); },
                    Ok(Cmd::SetAudioTrack(id)) => { let _ = mpv.set_property("aid", &id.to_string()); },
                    Ok(Cmd::SetSubtitleTrack(id)) => {
                        let v = if id < 0 { "no".to_owned() } else { id.to_string() };
                        let _ = mpv.set_property("sid", &v);
                    },
                    Ok(Cmd::SetWindowSize(w, h)) => { rctx.resize(w, h); },
                    Ok(Cmd::Quit) => {
                        info!("player thread quit requested");
                        return;
                    },
                    Err(tokio::sync::mpsc::error::TryRecvError::Empty) => break,
                    Err(tokio::sync::mpsc::error::TryRecvError::Disconnected) => return,
                }
            }

            // Pump one mpv event with short timeout so we stay responsive.
            let evt = mpv.wait_event(0.005);
            let id = unsafe { (*evt).event_id };
            if id != mpv_sys::MPV_EVENT_NONE {
                debug!("mpv event id={}", id);
                if let Some(e) = unsafe { events::from_mpv(evt) } {
                    let _ = evt_tx.send(e);
                }
                if id == mpv_sys::MPV_EVENT_PROPERTY_CHANGE {
                    if let Some((n, v)) = unsafe { events::property_change(evt) } {
                        let _ = evt_tx.send(Event::PropertyChanged { name: n, value: v });
                    }
                }
                if id == mpv_sys::MPV_EVENT_SHUTDOWN { return; }
            }

            // Render if available.
            if rctx.has_new_frame() {
                if let Err(e) = rctx.render_into_frames() {
                    error!("render failed: {}", e);
                    let _ = evt_tx.send(Event::Error { msg: e.to_string() });
                    thread::sleep(Duration::from_millis(50));
                }
            } else {
                // Sleep very briefly to avoid busy loop when nothing's happening.
                thread::sleep(Duration::from_micros(500));
            }
        }
    })?;

    Ok(PlayerHandle {
        cmd_tx,
        evt_rx: parking_lot::Mutex::new(evt_rx),
        frames,
    })
}
```

```rust
use std::sync::Arc;
use tracing::warn;

pub struct TvApp {
    player: PlayerHandle,
    video_tex: Option<TextureHandle>,
    last_frame_version: u64,
    status: String,
}

impl TvApp {
    pub fn new(_cc: &eframe::CreationContext<'_>, player: PlayerHandle, initial_load: Option<String>) -> Self {
        if let Some(u) = initial_load {
            let _ = player.cmd_tx.send(Cmd::LoadUrl(u));
        }
        // Set up: every new frame, request repaint on the egui thread.
        let ctx = _cc.egui_ctx.clone();
        player.frames.set_new_frame_callback(move || ctx.request_repaint());
        Self {
            player,
            video_tex: None,
            last_frame_version: 0,
            status: "tvplayer".into(),
        }
    }

    fn drain_events(&mut self) {
        let mut rx = self.player.evt_rx.lock();
        while let Ok(evt) = rx.try_recv() {
            match evt {
                Event::FileLoaded         => self.status = "playing".into(),
                Event::PlaybackStarted    => {},
                Event::EndOfFile { reason } => self.status = format!("end: {}", reason),
                Event::Error { msg }      => { warn!("player error: {}", msg); self.status = format!("error: {}", msg); },
                Event::PropertyChanged { .. } => {},
            }
        }
    }

    fn update_video_texture(&mut self, ctx: &egui::Context) {
        let frame: RgbaFrame = self.player.frames.read();
        if frame.w == 0 || frame.h == 0 { return; }
        if frame.version == self.last_frame_version && self.video_tex.is_some() { return; }
        self.last_frame_version = frame.version;

        // mpv produces BGRA0; egui wants RGBA. Swap R↔B inline.
        let mut rgba = frame.data.clone();
        for px in rgba.chunks_exact_mut(4) {
            px.swap(0, 2);
            px[3] = 255;
        }
        let img = ColorImage::from_rgba_unmultiplied([frame.w as usize, frame.h as usize], &rgba);
        match self.video_tex.as_mut() {
            Some(t) => t.set(img, TextureOptions::LINEAR),
            None    => self.video_tex = Some(ctx.load_texture("video", img, TextureOptions::LINEAR)),
        }
    }

    fn forward_window_size(&self, ctx: &egui::Context) {
        let r = ctx.input(|i| i.viewport().inner_rect);
        if let Some(rect) = r {
            let w = rect.width().max(1.0) as u32;
            let h = rect.height().max(1.0) as u32;
            let _ = self.player.cmd_tx.send(Cmd::SetWindowSize(w, h));
        }
    }
}

impl eframe::App for TvApp {
    fn update(&mut self, ctx: &egui::Context, _frame: &mut eframe::Frame) {
```

```rust
    pub fn spawn_fetch(&self) {
        let inner = self.inner.clone();
        let portal = self.portal.clone();
        tokio::spawn(async move {
            match portal.fetch_catalog().await {
                Ok(c) => {
                    info!("catalog loaded: {} live, {} movies, {} series", c.live.len(), c.movies.len(), c.series.len());
                    *inner.write().await = Some(c);
                },
                Err(e) => warn!("catalog fetch failed: {}", e),
            }
        });
    }

    pub fn live_channels_blocking(&self) -> Vec<LiveChannel> {
        self.inner.try_read().ok().and_then(|g| g.as_ref().map(|c| c.live.clone())).unwrap_or_default()
    }

    pub fn search_items_blocking(&self) -> Vec<SearchItem> {
        let g = match self.inner.try_read() {
            Ok(g) => g,
            Err(_) => return Vec::new(),
        };
        let Some(c) = g.as_ref() else { return Vec::new(); };
        let mut v = Vec::with_capacity(c.live.len() + c.movies.len() + c.series.len());
        for x in &c.live   { v.push(SearchItem { id: x.stream_id, name: x.name.clone(), kind: ItemKind::Live }); }
        for x in &c.movies { v.push(SearchItem { id: x.stream_id, name: x.name.clone(), kind: ItemKind::Movie }); }
        for x in &c.series { v.push(SearchItem { id: x.series_id, name: x.name.clone(), kind: ItemKind::Series }); }
        v
    }

    pub fn search_blocking(&self, query: &str) -> Vec<SearchItem> {
        let items = self.search_items_blocking();
        rank(query, &items).into_iter().cloned().collect()
    }

    pub fn portal(&self) -> &Arc<dyn Portal> { &self.portal }
```

```rust
/// Resolve the next live channel index based on direction.
/// `current` is None if no channel is currently selected.
pub fn next_live_idx(current: Option<usize>, len: usize, delta: i32) -> Option<usize> {
    if len == 0 { return None; }
    let cur = current.unwrap_or(0) as i32;
    let next = (cur + delta).rem_euclid(len as i32);
    Some(next as usize)
}

pub fn zap_to_idx(idx: usize, catalog: &CatalogStore, cmd_tx: &UnboundedSender<Cmd>) -> Option<String> {
    let live = catalog.live_channels_blocking();
    if idx >= live.len() { return None; }
    let ch = &live[idx];
    let url = catalog.portal().live_stream_url(ch.stream_id);
    let _ = cmd_tx.send(Cmd::LoadUrl(url));
    info!("zap → {} (stream_id={})", ch.name, ch.stream_id);
    Some(ch.name.clone())
}

pub fn zap_to_stream_id(stream_id: i64, catalog: &CatalogStore, cmd_tx: &UnboundedSender<Cmd>) -> Option<String> {
    let live = catalog.live_channels_blocking();
    let ch = live.iter().find(|c| c.stream_id == stream_id)?;
    let url = catalog.portal().live_stream_url(ch.stream_id);
    let _ = cmd_tx.send(Cmd::LoadUrl(url));
    info!("zap (by id) → {} ({})", ch.name, stream_id);
    Some(ch.name.clone())
}

/// Resolve "news shortcut": find any live channel whose name contains the needle
/// (case-insensitive). For NPO use ["NPO 1", "NPO 2", "NPO 3"] etc.
pub fn find_live_by_name(catalog: &CatalogStore, needle: &str) -> Option<i64> {
    let live = catalog.live_channels_blocking();
    live.iter().find(|c| c.name.to_lowercase().contains(&needle.to_lowercase())).map(|c| c.stream_id)
}

pub fn npo_shortcut_id(catalog: &CatalogStore, ch_num: u8) -> Option<i64> {
    let needle = format!("NPO {}", ch_num);
    find_live_by_name(catalog, &needle)
}

pub fn toggle_favorite(stream_id: i64, name: &str, favorites: &mut Favorites) -> bool {
    favorites.toggle(stream_id, name);
```
